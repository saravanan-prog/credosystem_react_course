### useCallback

- `useCallback` in React is used to `memoize a function`
- so that React does not `recreate` the function on `every render`.

### Syntax

```       
const memoizedFunction = useCallback(() => {
   // function logic
}, [dependencies]);
```

### Simple Example

```

import React, { useState, useCallback } from "react";

function App() {

  const [count, setCount] = useState(0);

  const showMessage = useCallback(() => {
    console.log("Button Clicked");
  }, []);

  return (
    <div>
      <h1>{count}</h1>

      <button onClick={() => setCount(count + 1)}>
        Increment
      </button>

      <button onClick={showMessage}>
        Show Message
      </button>
    </div>
  );
}

export default App;

```
