# React + Axios (Simple & Practical Guide)

- Axios is a popular library used in React to make API requests (GET, POST, PUT, DELETE).
- It is easier than fetch() and provides better error handling, interceptors, and cleaner syntax

#### 1️⃣ Install Axios

```
npm install axios

```

#### 2️⃣ Basic GET Request in React

```

import { useEffect, useState } from "react";
import axios from "axios";

function Users() {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    fetchUsers();
  }, []);

  async function fetchUsers() {
      try {
        const response = await axios.get(
          "https://jsonplaceholder.typicode.com/users"
        );
        setUsers(response.data);
      } catch (error) {
        console.error(error.message);
      }
    }

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}


```

#### 4️⃣ POST Request Example

```
import axios from "axios";

function createUser() {
  axios.post("https://jsonplaceholder.typicode.com/users", {
    name: "Saravanan",
    email: "test@gmail.com"
  })
  .then(response => {
    console.log("User created:", response.data);
  })
  .catch(error => {
    console.error(error.message);
  });
}
```

#### Axios Instance (Best Practice)

```
import { useEffect, useState } from "react";
import axios from "axios";


const api = axios.create({
  baseURL: "https://jsonplaceholder.typicode.com",
  timeout: 5000,
});

function Users() {
  const [users, setUsers] = useState([]);
  const [error, setError] = useState("");

  useEffect(() => {
    api
      .get("/users")
      .then(res => setUsers(res.data))
      .catch(err => setError(err.message));
  }, []);


  return (


    {!error ?
        <ul>
            {users.map(u => (
                <li key={u.id}>{u.name}</li>
            ))}
        </ul>

    : 
        <button> Re try Fetch user </button>

    }

  );
}
```

#### 6️⃣ Why Axios is better than fetch?

| Feature         | fetch  | Axios     |
| --------------- | ------ | --------- |
| JSON parsing    | manual | automatic |
| Error handling  | tricky | easy      |
| Interceptors    | ❌     | ✅        |
| Timeout support | ❌     | ✅        |
| Cleaner syntax  | ❌     | ✅        |
