# What is Fetch API?
- Fetch API is a built-in JavaScript tool used to send requests to a server and get data (usually JSON).
- In simple words:
```
    Fetch = “Ask backend for data and get response”
```

#### Basic Syntax

```
fetch(url)
  .then(response => response.json())
  .then(data => console.log(data))
  .catch(error => console.log(error));
```

***fetch-api.service.js***
```
const BASEURL = "https://fakestoreapi.com/"

const fetchApicall = async (pathUrl,method,payload) => {

  if(method.toLowerCase() == "get" || toLowerCase() == "delete")
    const response = await fetch(BASEURL + '/' + pathUrl)
  else if(method.toLowerCase() == "post" || method.toLowerCase() == "put")
    const response = await fetch(BASEURL + '/' + pathUrl, getOption(method,payload))
  
  const responseData = await response.json()
  return response
}


const getOption = (method,payload) => {
   return {
      method:method
      headers:{
        "Content-Type":"application/json"
      },
      body: JSON.strigyfy(payload)
   }
}

```
